### 0.2 `spike_w2/eta_v2.py` — 统一在线推断引擎（新增，旧入口保留）

```python
# 物理常量（单一事实源 = 本模块，W3 import；值与 Doc14 §5 代码逐字一致）
UNIVERSAL_EXIT_STATIONS: frozenset[str]   # 37 站（Doc14 §5 逐字）
BLACKTOP_ONLINE_WHITELIST: frozenset[str] # 10 站（报交入口簇 6 + 打磨返修线 4，含 14IS120）
SPRAY_QUEUE_BLACKLIST: frozenset[str]     # {'14IS260', '13IS260'}
OVEN_OUT_STATIONS: frozenset[str]         # {'16IS305','17IS305','18IS345'}  # 3 站（Doc14 勘误：20IS010 移出=2 线储存线入口）

def resolve_effective_flag(station: str, flag137: str, flag139: str,
                           passed_oven: bool) -> str:
    """Doc14 §5 状态机逐字：37 站强制解冻 '0' → 139=2 三重防护（137=1 / 喷房队列抑制 /
    未出烘房须白名单）→ 139=1 直通 → 其余 '0'。纯函数。"""

def estimate_event_unified(t: int, station: str, flag137: str, flag139: str,
                           model: str, week: str, tables: dict,
                           mode: str = 'vehicle', passed_oven: bool = False,
                           comp_excl: frozenset = frozenset(),
                           effective_flag: str | None = None) -> tuple:
    """统一路由（Doc14 §5 语义 + W2 既有 week/comp_excl 超集参数）。
    返回 (eta 绝对秒|None, used_cohort bool)，与旧入口同形状。

    eff = (effective_flag if effective_flag is not None
           else resolve_effective_flag(station, flag137, flag139, passed_oven))
    # ⚠️ effective_flag = 离线净化 flag 直通通道（agy 阻断项 3 修复）：
    #   02_replay_shift 回放端 / 单测端传 tmp_ev_base 已净化 flag（'0'/'1'/'2'），跳过在线状态机——
    #   离线净化打标即真值，在线状态机是其无状态镜像；若回放端改传 raw 137/139 + passed_oven，
    #   缺 137 位会把纯净黑顶（fl='2'）全部误杀回 '0'，G2 门判失真。在线端（predictor）恒传 None 走全状态机。
    分支 A（eff ∈ {'1','2'}）：
        rem = resolve_hierarchical(tables['loop_rem'], (station, eff), model, mode)
            # 车型 → 平台前缀 → ALL，n≥30；消费 p50(vehicle)/mean(shift)
        rem 缺 → resolve_hierarchical(tables['loop_dur'], (eff,), model, mode)  # L 兜底
        eta = t + round(rem)                      # C8：全旅程语义，不叠加任何项
    分支 B（eff == '0'）：
        base = resolve_cohort_p50(...) or resolve_hierarchical(tables['rem'], station, model, mode)
        base 缺 → (None, False)
        delay = (0.0 if station in UNIVERSAL_EXIT_STATIONS
                 else resolve_hierarchical(tables['expected_delay'], (station,), model, mode))
                 # 新表替代旧 p×L1+p×L2（doc10 合流日语义）；出网关后恒 0
        eta = t + round(base + delay)
    """
```

行形状扩展（tables 装载）：
```text
rem:          {(station, model): (p50, mean, p90, n)}          # 不变（model 含前缀行 + 'ALL'）
loop_rem:     {(station, flag, model): (p50, mean, p90, n)}    # 不变
loop_dur:     {(flag, model): (l, l_mean, l_p90, n)}           # 2 → 4 元组（C11；n 仍在末位、l 仍在首位，
                                                               # spike_w1/eta.py resolve 取 row[-1]=n / row[0]=l → 零改动）
expected_delay: {(station, model): (delay_mean_s, delay_p50_s, n_fut, n_clean)}  # 新增
# ⚠️ expected_delay 列序与 rem/loop_rem/loop_dur 相反（mean 在前、p50 第二）：
#   解析必须显式索引（shift → row[0] mean；vehicle → row[1] p50），禁走 row[0] 通用取值，否则 mean/p50 颠倒；
#   回退判定 = 双人群双门槛（n_fut≥30 且 n_clean≥30 各自达标方可用该行，缺一降层），独立于其余表 n≥30 通用门槛。
#   专用解析器 `resolve_expected_delay(tables, station, model, mode) -> int | None`（双审 agy 建议 3，W4.3a 实现）：
#   封装 车型行→前缀行→缺行 0 三层回退 + 双门槛判定 + 显式列索引；统一引擎 / predictor / dashboard 三处调用点必须走此函数，禁止行内解析；
#   旧引擎两站点（eta.py / eta_v2.py resolve_event）不动（零改动红线，旧路径不消费本表除非 DSS_USE_NET_DELAY=1）。
fut_week/fut_all/cohort/lineprob/quality: 不变
```
