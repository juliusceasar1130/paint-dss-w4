# spike_w3/predictor.py（时区纪律同 T2：全 aware Asia/Shanghai；naive 抛 TypeError）

def rem_of(v: VehicleState, t: datetime, tables: dict, comp_excl: frozenset) -> Optional[float]:
    """单车模型剩余秒数（t 无关：= eta_v2.estimate_event(last_event_ts, station, flag,
    body_type, iso_week_label(t.date()), tables, comp_excl) 的 ETA − last_event_ts）。
    查不到分布（主/环表缺行 n<30）→ None（no-ETA 桶，报告可见化，不计 pred）。
    week 标签用锚点 t 的 ISO 周（同 W2 静态：iso_week_label(t_dt.date())）。"""

def a_of(conn, shift: shifts.Shift, t: datetime) -> int:
    """A(t) = [shift.attr_start, t] 内下线产品车数（4 出口站并集、按出口时刻归班）。
    区间语义 = **闭区间 [attr_start, t] 两端含**（与 W2 静态完全同义：exit_arr >= attr_ep
    & exit_arr <= t，02_replay_shift.py:260）→ 恰在 16:30:00 的出口事件日班（t=end 含）与
    中班（attr_start 含）都计——与静态行为一致（构造性一致）；单测固化交接边界用例
    （外部审低⑤）。来源 = ods 有界 SQL（窗内任一出口事件即计该车一次 =
    COUNT(DISTINCT BODY_ID)，占位符 2 个），**不用池**（池 R3 出站即移除，无出站时刻戳；
    SQL = W2 静态 exit_arr 同源 → α=off 一致性由构造保证）。
    窗 ≤ ~12h，有界、快（2026-09-25 外部审真库实测单次 12ms）；M9 每锚点 1 次，在线每轮 1 次。"""

def predict(conn, pool: VehiclePool, shift: shifts.Shift, t: datetime,
            tables: dict, comp_excl: frozenset, cfg: DSSConfig,
            alpha: float = 1.0) -> dict:
    """pred(t) = A(t) + Σ [t + rem_v/alpha ≤ shift.end]（W2-v3 口径 rem' = rem/α）。
    在制集 = pool.snapshot() ∩ 产品车（body_id 以 78202 起头——池契约无过滤，
    预测层过滤与 W2 静态 LIKE '78202%' 同口径；池构造已排除出站/延续站车）。
    返回 {shift_id, t, A, inproc, pred, no_eta, stale,
          by_vehicle: [{body_id, station, flag, body_type, rem_sec, stale: bool}]}。
    L4：stale = (t − last_event_ts) > eta_stale_hours → 明细标 stale=True，**仍计 pred**
    （α=off 与静态一致性要求：静态无 stale 剔除）。"""

def build_takt_baseline(conn, win_start: str, win_end: str,
                        min_samples: int) -> dict:
    """α 分母：历史 P50 逐小时出站速率 × 班次型（docs/02 C1 定稿：启动时 SQL 聚合、
    缓存 dict、启动报告打印生效值；不新增表）。
    步骤：① 学习窗逐日类型 = shifts.detect_day_type/detect_mid_type（复用 W2
    SQL_DAYTYPE/SQL_MIDLAST，importlib 只读引 02_replay_shift 模块级常量；
    对**纯历史窗**使用，无未来泄漏——T1.3 禁用是针对在线因果兜底，不覆盖本场景，R4）；
    ② 窗内产品车出口事件（4 站）按出口时刻归班（04:45–16:30 → 当日日班型；
    16:30–04:45 → 当日中班型；类型 None 的天剔除）；③ 按 (kind, type) 小时桶计数
    → 逐小时速率 → P50；④ n_hours < min_samples → 回退最近班次型
    （按工时距离 8h→10h→11h，同 kind；再不足 → 该型无基准 α 恒 1.0 + 标注）。
    返回 {(kind, type): (p50_rate: float, n_hours: int, source: str)}。"""

class AlphaTracker:
    """α 状态机（B5 定稿参数；纯状态，无 DB 依赖 → 单测可全合成驱动）。
    参数（构造注入，默认取 cfg）：window_hours=3.0 / min_samples=30 / ema_beta=0.3 /
    clamp [0.6, 1.5]；迟滞 0.05 + 2 周期 = 代码级常量（2026-09-23 复审 B1 修正，不入 env）。
    - baseline: dict（build_takt_baseline 产物；缺该型 → α 恒 1.0 + 标注 no_baseline）
    - on_shift(shift)：班次切换（新 attr_start 到达）→ committed=1.0、ema=None、
      迟滞计数清零、绑定 (kind, type) 基准。
    - update(t, shift, exits_in_window: int, window_hours: float) -> float：
      t ≥ shift.end 或班次外 → 无观察（返回当前 committed，状态静止，外部审低⑥）；
      观察窗 = [max(t−window, attr_start), min(t, end−30min)]（尾 30min 剔除 = B5 定稿
      「收班前 30min 尾部」；生产 Phase 1 停线剔除集 = 空集，T5 接入后传入）；
      窗长 < 1h → 无观察（返回当前 committed，不动状态）；
      α_raw = (exits/window_hours) / p50_rate → **即时 clamp raw_clamped ∈ [0.6, 1.5]**
      （防单尖峰无界推高 ema，外部审中③-1）；
      ema = β·raw_clamped + (1−β)·ema_prev（首观 = raw_clamped）；
      **同向迟滞**：(ema − committed) 同号且 |Δ| ≥ 0.05 连续 2 周期 → committed = ema
      （已 clamp 界内；+0.06/−0.06 振荡不提交，外部审中③-2）；否则保持。
      返回 committed。α 放大语义：α>1 = 比历史快 → rem/α 缩短 → pred 增大。"""
