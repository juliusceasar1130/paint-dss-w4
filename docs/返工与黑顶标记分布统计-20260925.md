核心 SQL（首现分布，经 `common.connect()` 连接，会话时区已内置 `Asia/Shanghai`）：

```sql
WITH e AS (
    SELECT "BODY_ID" AS body, "RW_STATION_ID" AS station, "DATE_EVT" AS t,
           SUBSTRING("MDS_DATA" FROM 139 FOR 1) AS flag
    FROM ods.carbody_history
    WHERE "BODY_ID" LIKE '78202%'
      AND SUBSTRING("MDS_DATA" FROM 139 FOR 1) IN ('1','2')
),
f AS (
    SELECT body, station, flag,
           ROW_NUMBER() OVER (PARTITION BY body, flag ORDER BY t) AS rn
    FROM e
)
SELECT flag, station, count(*) AS n
FROM f WHERE rn = 1
GROUP BY flag, station
ORDER BY flag, n DESC;
```
